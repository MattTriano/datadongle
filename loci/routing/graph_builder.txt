"""
SafetyWeightedGraphBuilder — builds a NetworkX routing graph from the
bike_safety_weighted_edges table with tunable composite weights.

Usage:
    from loci.routing.graph_builder import SafetyWeightedGraphBuilder

    builder = SafetyWeightedGraphBuilder(engine=postgis_engine)
    G = builder.build()
    builder.save(G, "/tmp/bike_routing_graph.pickle")

    # Or as a one-liner:
    G = builder.build_and_save("/tmp/bike_routing_graph.pickle")

    # With custom weight tuning:
    config = BikeSafetyFactorConfig(crash_alpha=3.0)
    builder = SafetyWeightedGraphBuilder(engine=postgis_engine, config=config)
    G = builder.build()
"""

import logging
import pickle
from dataclasses import dataclass, field
from pathlib import Path

import networkx as nx


logger = logging.getLogger(__name__)

# Schemas and table names
# WEIGHTED_EDGES_TABLE = "dbt_loci_marts.bike_safety_weighted_edges"
# NODES_TABLE = "raw_data.osmnx_bike_network_nodes"


@dataclass
class BikeSafetyFactorConfig:
    """Tunable parameters for computing safety-weighted edge costs.
 
    Attributes
    ----------
    road_type_multipliers : dict[str, float]
        Multiplier by OSM highway type. Higher = less preferred.
        Applied to all edges based on their highway tag.
    infra_factors : dict[int, float]
        Multiplier by infrastructure quality tier (1-10).
        Lower tier = better infrastructure = lower factor.
    crash_alpha : float
        Scaling factor for crash score contribution.
        Higher = crashes penalize the edge more heavily.
        The crash factor is: 1 + crash_alpha * crash_score_per_meter.
    default_road_type_multiplier : float
        Fallback multiplier for highway types not in the dict.
    """
 
    road_type_multipliers: dict[str, float] = field(default_factory=lambda: {
        "cycleway": 0.5,
        "path": 0.7,
        "living_street": 0.7,
        "residential": 0.8,
        "tertiary": 1.0,
        "tertiary_link": 1.0,
        "secondary": 1.5,
        "secondary_link": 1.5,
        "primary": 2.0,
        "primary_link": 2.0,
        "trunk": 3.0,
        "trunk_link": 3.0,
        "motorway": 5.0,
        "motorway_link": 5.0,
        "unclassified": 1.0,
        "service": 0.9,
        "pedestrian": 0.8,
        "footway": 0.9,
        "steps": 2.0,
    })
 
    infra_factors: dict[int, float] = field(default_factory=lambda: {
        1: 0.4,    # dedicated cycleway
        2: 0.5,    # protected lane
        3: 0.6,    # dedicated shared path / footway
        4: 0.65,   # buffered sharrow
        5: 0.7,    # neighborhood greenway
        6: 0.85,   # shoulder / shared busway / other
        7: 0.75,   # buffered lane
        8: 0.8,    # unbuffered bike lane
        9: 0.9,    # unbuffered sharrow
        10: 1.0,   # no infrastructure
    })
 
    crash_alpha: float = 2.0

    default_road_type_multiplier: float = 1.0

class SafetyWeightedGraphBuilder:
    """Builds a safety-weighted NetworkX graph for bike routing.

    Reads the enriched edge table (with infra quality tiers and crash
    scores) and the node table from PostGIS, reconstructs a NetworkX
    MultiDiGraph, and computes a composite routing weight on each edge.

    Parameters
    ----------
    engine : PostgresEngine
    config : BikeSafetyFactorConfig, optional
        Tunable weight parameters. Uses defaults if not provided.
    edges_table : str, optional
        Fully qualified table name for weighted edges.
    nodes_table : str, optional
        Fully qualified table name for network nodes.
    """

    def __init__(
        self,
        engine,
        edges_table: str,
        nodes_table: str,
        config: BikeSafetyFactorConfig | None = None,
    ):
        self.engine = engine
        self.config = config or BikeSafetyFactorConfig()
        self.edges_table = edges_table
        self.nodes_table = nodes_table

    # ------------------------------------------------------------------
    # Public API
    # ------------------------------------------------------------------

    def build(self) -> nx.MultiDiGraph:
        """Read from PostGIS and build the weighted graph.

        Returns
        -------
        networkx.MultiDiGraph
            Graph with a 'safety_weight' attribute on each edge.
        """
        logger.info("Loading nodes from %s", self.nodes_table)
        nodes = self._load_nodes()
        logger.info("Loaded %d nodes", len(nodes))

        logger.info("Loading edges from %s", self.edges_table)
        edges = self._load_edges()
        logger.info("Loaded %d edges", len(edges))

        logger.info("Building graph")
        G = self._build_graph(nodes, edges)
        logger.info(
            "Graph built: %d nodes, %d edges",
            G.number_of_nodes(),
            G.number_of_edges(),
        )

        logger.info("Computing safety weights")
        self._compute_weights(G)
        logger.info("Safety weights computed")

        return G

    def save(self, graph: nx.MultiDiGraph, path: str | Path) -> Path:
        """Serialize graph to a pickle file.

        Parameters
        ----------
        graph : networkx.MultiDiGraph
        path : str or Path
            Output file path. Parent directories are created if needed.

        Returns
        -------
        Path to the saved file.
        """
        path = Path(path)
        path.parent.mkdir(parents=True, exist_ok=True)

        with open(path, "wb") as f:
            pickle.dump(graph, f, protocol=pickle.HIGHEST_PROTOCOL)

        size_mb = path.stat().st_size / (1024 * 1024)
        logger.info("Saved graph to %s (%.1f MB)", path, size_mb)
        return path

    def build_and_save(self, path: str | Path) -> nx.MultiDiGraph:
        """Build the graph and save it. Returns the graph."""
        G = self.build()
        self.save(G, path)
        return G

    @staticmethod
    def load(path: str | Path) -> nx.MultiDiGraph:
        """Load a previously saved graph from a pickle file."""
        path = Path(path)
        with open(path, "rb") as f:
            G = pickle.load(f)
        logger.info(
            "Loaded graph from %s: %d nodes, %d edges",
            path,
            G.number_of_nodes(),
            G.number_of_edges(),
        )
        return G

    # ------------------------------------------------------------------
    # Data loading
    # ------------------------------------------------------------------

    def _load_nodes(self) -> list[dict]:
        """Load current network nodes from PostGIS."""
        return self.engine.query(
            f"""
            SELECT osmid, latitude, longitude
            FROM {self.nodes_table}
            WHERE valid_to IS NULL
            """,
            as_dicts=True,
        )

    def _load_edges(self) -> list[dict]:
        """Load weighted edges from PostGIS.

        Selects only the columns needed for graph construction and
        weight computation — skips the geometry column to save memory.
        """
        return self.engine.query(
            f"""
            SELECT
                u, v, key, osmid, name, highway, oneway, length_m,
                maxspeed, surface, lanes, lit, bridge, tunnel,
                matched_infra_type, matched_infra_category,
                quality_tier, matched_has_buffer,
                crash_count, crash_score, crash_score_per_meter
            FROM {self.edges_table}
            """,
            as_dicts=True,
        )

    # ------------------------------------------------------------------
    # Graph construction
    # ------------------------------------------------------------------

    def _build_graph(
        self,
        nodes: list[dict],
        edges: list[dict],
    ) -> nx.MultiDiGraph:
        """Construct a NetworkX MultiDiGraph from nodes and edges."""
        G = nx.MultiDiGraph()

        for node in nodes:
            G.add_node(
                node["osmid"],
                x=node["longitude"],
                y=node["latitude"],
            )

        for edge in edges:
            u = edge["u"]
            v = edge["v"]
            key = edge["key"]

            attrs = {
                "osmid": edge["osmid"],
                "name": edge["name"],
                "highway": edge["highway"],
                "oneway": edge["oneway"],
                "length": edge["length_m"],
                "maxspeed": edge["maxspeed"],
                "surface": edge["surface"],
                "lanes": edge["lanes"],
                "lit": edge["lit"],
                "bridge": edge["bridge"],
                "tunnel": edge["tunnel"],
                "matched_infra_type": edge["matched_infra_type"],
                "matched_infra_category": edge["matched_infra_category"],
                "quality_tier": edge["quality_tier"],
                "matched_has_buffer": edge["matched_has_buffer"],
                "crash_count": edge["crash_count"],
                "crash_score": edge["crash_score"],
                "crash_score_per_meter": edge["crash_score_per_meter"],
            }

            G.add_edge(u, v, key=key, **attrs)

        return G

    # ------------------------------------------------------------------
    # Weight computation
    # ------------------------------------------------------------------

    def _compute_weights(self, G: nx.MultiDiGraph) -> None:
        """Compute and attach composite safety weights to all edges.

        The weight formula:
            weight = length * road_type_mult * infra_factor * crash_factor

        Where:
            road_type_mult: penalty for busy roads (from highway tag)
            infra_factor:   bonus for good bike infrastructure (from quality_tier)
            crash_factor:   penalty for crash history (1 + alpha * crash_score_per_meter)

        Each component is also stored on the edge for inspection.
        """
        config = self.config

        for u, v, key, data in G.edges(data=True, keys=True):
            length = data.get("length", 100)

            # Road type multiplier
            highway = data.get("highway", "")
            if isinstance(highway, list):
                highway = highway[0]
            road_type_mult = config.road_type_multipliers.get(
                highway, config.default_road_type_multiplier
            )

            # Infrastructure factor
            quality_tier = data.get("quality_tier", 10)
            infra_factor = config.infra_factors.get(quality_tier, 1.0)

            # Crash factor
            crash_score_per_meter = data.get("crash_score_per_meter", 0)
            crash_factor = 1.0 + config.crash_alpha * crash_score_per_meter

            # Composite weight
            weight = length * road_type_mult * infra_factor * crash_factor

            # Store everything for debuggability
            data["road_type_mult"] = road_type_mult
            data["infra_factor"] = infra_factor
            data["crash_factor"] = crash_factor
            data["safety_weight"] = weight
